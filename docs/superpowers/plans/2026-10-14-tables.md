# โต๊ะ (Tables) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** เจ้าของวางผังโต๊ะจริงของสาขาได้ แคชเชียร์เห็นสถานะโต๊ะ เปิดออเดอร์ที่โต๊ะ ย้ายโต๊ะ รวมบิล และแยกบิลตามรายการได้

**Architecture:** โมดูล API ใหม่ `tables` (ต้นแบบ `categories`) เก็บโต๊ะของสาขาและตำแหน่งบนผัง ออเดอร์ชี้โต๊ะผ่าน `orders.table_id` คำสั่งย้าย/รวม/แยกเป็นเมธอดของ `OrderEntity` และ use case ในโมดูล `orders` ที่ล็อกแถวตามลำดับ id สถานะโต๊ะ (ว่าง/มีลูกค้า) คำนวณจากออเดอร์เปิดผ่าน port `ITableOccupancy` ที่โมดูล `orders` เป็นผู้ implement ฝั่งเว็บเพิ่ม feature `tables` (ตัวแก้ผังของเจ้าของ และคอมโพเนนต์ผังที่แสดงอย่างเดียว) และเครื่องมือโต๊ะในหน้า POS

**Tech Stack:** Bun, Elysia, Drizzle (PostgreSQL), tsyringe, zod · React 19, TanStack Query, zustand, i18next · Playwright (`bun run e2e`)

**Spec:** `docs/superpowers/specs/2026-10-14-tables-design.md`

กรณีที่ spec ไม่ได้พูดตรง ๆ แต่ผู้ใช้จริงจะเจอ (แต่ละข้อมีการทดสอบกำกับใน task ที่ระบุ)

```ts
import { DomainError } from './errors/domain-error'

/** The floor plan is drawn in these units, whatever size the screen is. */
export const LAYOUT_WIDTH = 1000
export const LAYOUT_HEIGHT = 700
export const MIN_TABLE_SIZE = 40
export const MAX_SEATS = 99
const MAX_TEXT_LENGTH = 50

const invalid = (message: string) => new DomainError(message, 'INVALID_TABLE')
const isWhole = (value: number) => Number.isInteger(value)
const definedOnly = <T extends object>(changes: T) =>
  Object.fromEntries(Object.entries(changes).filter(([, value]) => value !== undefined)) as T

export interface DiningTableProps {
  /** 0 until the table has been stored. */
  id: number
  branchId: number
  name: string
  zone: string | null
  seats: number
  /** Position and size on the floor plan, in layout units. */
  x: number
  y: number
  width: number
  height: number
  sortOrder: number
  isActive: boolean
}

export class DiningTableEntity {
  public readonly id: number
  public readonly branchId: number
  public readonly name: string
  public readonly zone: string | null
  public readonly seats: number
  public readonly x: number
  public readonly y: number
  public readonly width: number
  public readonly height: number
  public readonly sortOrder: number
  public readonly isActive: boolean

  constructor(props: DiningTableProps) {
    if (props.name.length < 1 || props.name.length > MAX_TEXT_LENGTH)
      throw invalid(`ชื่อโต๊ะต้องยาว 1 ถึง ${MAX_TEXT_LENGTH} ตัวอักษร`)
    if (props.zone !== null && (props.zone.length < 1 || props.zone.length > MAX_TEXT_LENGTH))
      throw invalid(`ชื่อโซนต้องยาว 1 ถึง ${MAX_TEXT_LENGTH} ตัวอักษร`)
    if (!isWhole(props.seats) || props.seats < 1 || props.seats > MAX_SEATS)
      throw invalid(`จำนวนที่นั่งต้องเป็นจำนวนเต็ม 1 ถึง ${MAX_SEATS}`)
    if (![props.x, props.y, props.width, props.height].every(isWhole)) throw invalid('ตำแหน่งและขนาดของโต๊ะต้องเป็นจำนวนเต็ม')
    if (props.width < MIN_TABLE_SIZE || props.height < MIN_TABLE_SIZE)
      throw invalid(`โต๊ะต้องกว้างและสูงอย่างน้อย ${MIN_TABLE_SIZE}`)
    if (props.x < 0 || props.y < 0 || props.x + props.width > LAYOUT_WIDTH || props.y + props.height > LAYOUT_HEIGHT)
      throw invalid('โต๊ะต้องอยู่ในพื้นที่ของผัง')
    if (!isWhole(props.sortOrder)) throw invalid('ลำดับการแสดงผลต้องเป็นจำนวนเต็ม')

    this.id = props.id
    this.branchId = props.branchId
    this.name = props.name
    this.zone = props.zone
    this.seats = props.seats
    this.x = props.x
    this.y = props.y
    this.width = props.width
    this.height = props.height
    this.sortOrder = props.sortOrder
    this.isActive = props.isActive
  }

  /** A copy with the given fields changed; undefined fields keep their current value. */
  with(changes: Partial<Omit<DiningTableProps, 'id' | 'branchId'>>): DiningTableEntity {
    return new DiningTableEntity({ ...this.toSnapshot(), ...definedOnly(changes) })
  }

  toSnapshot(): DiningTableProps {
    return {
      id: this.id,
      branchId: this.branchId,
      name: this.name,
      zone: this.zone,
      seats: this.seats,
      x: this.x,
      y: this.y,
      width: this.width,
      height: this.height,
      sortOrder: this.sortOrder,
      isActive: this.isActive,
    }
  }
}
```

```ts
import { boolean, index, integer, pgTable, text, unique } from 'drizzle-orm/pg-core'
import { id, timestamps } from '../columns'
import { branches } from './branches'

// A table on the floor plan of a branch. Its position and size are in layout units (see the domain).
export const diningTables = pgTable(
  'dining_tables',
  {
    id: id(),
    branchId: integer()
      .notNull()
      .references(() => branches.id),
    name: text().notNull(),
    zone: text(),
    seats: integer().notNull().default(4),
    x: integer().notNull().default(20),
    y: integer().notNull().default(20),
    width: integer().notNull().default(120),
    height: integer().notNull().default(80),
    sortOrder: integer().notNull().default(0),
    isActive: boolean().notNull().default(true),
    ...timestamps,
  },
  (table) => [unique().on(table.branchId, table.name), index().on(table.branchId)],
)
```

```ts
  /** Puts the order on another table, or on none. */
  moveToTable(tableId: number | null): OrderEntity {
    this.assertOpen()
    return this.change({ tableId })
  }

  /**
   * Takes the lines of another open order onto this one. A line of the same product and options adds
   * to this order's line and keeps the price quoted here. Call markMerged on the source afterwards.
   */
  absorb(source: OrderEntity): OrderEntity {
    this.assertOpen()
    source.assertOpen()
    if (source.id === this.id) throw new DomainError('รวมออเดอร์เข้ากับตัวเองไม่ได้', 'ORDER_MERGE_SAME')
    return this.change({ items: mergeLines([...this.items, ...source.items.map((item) => ({ ...item, lineId: 0 }))]) })
  }

  /** Closes this order as merged into another: its lines have moved there. */
  markMerged(intoId: number): OrderEntity {
    this.assertOpen()
    return this.change({ status: 'merged', mergedIntoId: intoId, items: [] })
  }

  /**
   * Takes part of the lines off this order into a new order on the same table. At least one unit must
   * stay here. Prices and options go with the lines as quoted.
   */
  splitOff(
    selection: readonly SplitSelection[],
    actor: { id: number; displayName: string },
    label: string | null,
  ): { remaining: OrderEntity; created: NewOrder } {
    this.assertOpen()
    if (selection.length === 0) throw new DomainError('ต้องเลือกรายการที่จะแยกอย่างน้อยหนึ่งรายการ', 'ORDER_SPLIT_EMPTY')
    if (new Set(selection.map((pick) => pick.lineId)).size !== selection.length) throw invalid('เลือกบรรทัดเดียวกันซ้ำไม่ได้')
    for (const pick of selection) {
      assertQuantity(pick.quantity)
      if (!this.items.some((item) => item.lineId === pick.lineId))
        throw new DomainNotFoundError('ไม่พบสินค้านี้ในออเดอร์', 'ORDER_ITEM_NOT_FOUND')
    }

    const moved: OrderItem[] = []
    const kept = this.items.flatMap((item) => {
      const pick = selection.find((candidate) => candidate.lineId === item.lineId)
      if (!pick) return [item]
      if (pick.quantity > item.quantity)
        throw new DomainError('จำนวนที่แยกมากกว่าที่มีในบรรทัด', 'ORDER_SPLIT_QUANTITY', { available: item.quantity })
      moved.push({ ...item, lineId: 0, quantity: pick.quantity })
      return pick.quantity === item.quantity ? [] : [{ ...item, quantity: item.quantity - pick.quantity }]
    })
    if (kept.length === 0) throw new DomainError('ต้องเหลืออย่างน้อยหนึ่งรายการในออเดอร์เดิม', 'ORDER_SPLIT_EMPTY')

    return {
      remaining: this.change({ items: kept }),
      created: OrderEntity.open({
        branchId: this.branchId,
        label: label ?? this.label,
        tableId: this.tableId,
        openedById: actor.id,
        openedByName: actor.displayName,
        items: moved,
      }),
    }
  }
```

```ts
      .from(diningTables)
      .where(eq(diningTables.branchId, branchId))
      .orderBy(diningTables.sortOrder, diningTables.name)

      return new DiningTableEntity(row!)
    } catch (error) {
      if (isUniqueViolation(error)) throw duplicate()
      throw error
    }
  }

  async update(entity: DiningTableEntity) {

    try {
      const [row] = await currentExecutor().update(diningTables).set(values).where(eq(diningTables.id, id)).returning(columns)
```

```ts
import { and, count, eq, isNotNull, sql } from 'drizzle-orm'
import { injectable } from 'tsyringe'
import { orderItems, orders } from '../../../../infrastructure/database/schema/orders'
import { currentExecutor } from '../../../../libs/transaction-context'
import type { ITableOccupancy, TableOccupancy } from '../../../tables/applications/ports/table-occupancy.port'

/** Answers the tables module's question from the orders' own data, so it never reads our tables. */
@injectable()
export class DrizzleTableOccupancy implements ITableOccupancy {
  async byTable(branchId: number) {
    const rows = await currentExecutor()
      .select({
        tableId: orders.tableId,
        openOrders: sql<number>`count(distinct ${orders.id})`.mapWith(Number),
        openTotalSatang: sql<number>`coalesce(sum(${orderItems.unitPriceSatang} * ${orderItems.quantity}), 0)`.mapWith(Number),
      })
      .from(orders)
      .leftJoin(orderItems, eq(orderItems.orderId, orders.id))
      .where(and(eq(orders.branchId, branchId), eq(orders.status, 'open'), isNotNull(orders.tableId)))
      .groupBy(orders.tableId)

    return new Map<number, TableOccupancy>(
      rows.map((row) => [row.tableId!, { openOrders: row.openOrders, openTotalSatang: row.openTotalSatang }]),
    )
  }

  async openOrderCount(tableId: number) {
    const [row] = await currentExecutor()
      .select({ total: count() })
      .from(orders)
      .where(and(eq(orders.tableId, tableId), eq(orders.status, 'open')))
    return row!.total
  }
}
```

```ts
import { inject, injectable } from 'tsyringe'
import { DomainError } from '../../../../domains/errors/domain-error'
import type { IUnitOfWork } from '../../../shared/ports/unit-of-work.port'
import { TOKENS } from '../../../tokens'
import { orderRepositoryToken } from '../ports/order.repository'
import type { IOrderRepository } from '../ports/order.repository'
import { mergeOrdersSchema } from '../schemas/order-schemas'
import { inBranch } from '../services/order-access'

/**
 * Command: takes the lines of another open order onto this one and closes that one as merged. Both
 * rows are locked in id order, so two tills merging the orders in opposite directions cannot deadlock.
 */
@injectable()
export class MergeOrdersUseCase {
  constructor(
    @inject(orderRepositoryToken) private readonly orders: IOrderRepository,
    @inject(TOKENS.UnitOfWork) private readonly unitOfWork: IUnitOfWork,
  ) {}

  async execute(id: number, input: unknown, branchId: number) {
    const { sourceOrderId } = mergeOrdersSchema.parse(input)
    if (sourceOrderId === id) throw new DomainError('รวมออเดอร์เข้ากับตัวเองไม่ได้', 'ORDER_MERGE_SAME')

    return this.unitOfWork.run(async () => {
      const [first, second] = id < sourceOrderId ? [id, sourceOrderId] : [sourceOrderId, id]
      const lockedFirst = inBranch(await this.orders.findByIdForUpdate(first), branchId)
      const lockedSecond = inBranch(await this.orders.findByIdForUpdate(second), branchId)
      const destination = lockedFirst.id === id ? lockedFirst : lockedSecond
      const source = lockedFirst.id === id ? lockedSecond : lockedFirst

      const merged = destination.absorb(source)
      await this.orders.save(source.markMerged(destination.id))
      return this.orders.save(merged)
    })
  }
}
```

```ts
import { LAYOUT_HEIGHT, LAYOUT_WIDTH, MIN_TABLE_SIZE } from '@pos/api/dining-table'

export interface Rect {
  x: number
  y: number
  width: number
  height: number
}

const clamp = (value: number, min: number, max: number) => Math.min(Math.max(value, min), max)

/** A distance on a canvas of `canvasPixels` across a layout of `layoutUnits`, in whole layout units. */
export const pixelsToUnits = (pixels: number, canvasPixels: number, layoutUnits: number) =>
  Math.round((pixels * layoutUnits) / canvasPixels)

/** The table moved by a distance, held inside the floor. */
export function moveRect(rect: Rect, dx: number, dy: number): Rect {
  return {
    ...rect,
    x: clamp(Math.round(rect.x + dx), 0, LAYOUT_WIDTH - rect.width),
    y: clamp(Math.round(rect.y + dy), 0, LAYOUT_HEIGHT - rect.height),
  }
}

/** The table resized from its bottom right corner, never below the minimum and never off the floor. */
export function resizeRect(rect: Rect, dWidth: number, dHeight: number): Rect {
  return {
    ...rect,
    width: clamp(Math.round(rect.width + dWidth), MIN_TABLE_SIZE, LAYOUT_WIDTH - rect.x),
    height: clamp(Math.round(rect.height + dHeight), MIN_TABLE_SIZE, LAYOUT_HEIGHT - rect.y),
  }
}
```

```tsx
export function TablesPage() {
  const { t } = useTranslation('tables')
  const { t: tc } = useTranslation()
  const tables = useQuery(tablesQuery)
  const update = useUpdateTable()
  const [adding, setAdding] = useState(false)
  const [selectedId, setSelectedId] = useState<number | null>(null)
  const [drag, setDrag] = useState<Drag>(null)
  // While dragging, the table is drawn at the draft place; it is saved when the pointer is released.
  const [draft, setDraft] = useState<(Rect & { id: number }) | null>(null)
  const canvas = useRef<HTMLDivElement>(null)

  const all = tables.data ?? []
  const selected = all.find((table) => table.id === selectedId) ?? null

  const begin = (event: PointerEvent, table: DiningTable, mode: 'move' | 'resize') => {
    event.stopPropagation()
    event.currentTarget.setPointerCapture(event.pointerId)
    setSelectedId(table.id)
    setDrag({
      id: table.id,
      mode,
      startX: event.clientX,
      startY: event.clientY,
      origin: { x: table.x, y: table.y, width: table.width, height: table.height },
    })
  }

  const track = (event: PointerEvent) => {
    if (!drag || !canvas.current) return
    const box = canvas.current.getBoundingClientRect()
    const dx = pixelsToUnits(event.clientX - drag.startX, box.width, LAYOUT_WIDTH)
    const dy = pixelsToUnits(event.clientY - drag.startY, box.height, LAYOUT_HEIGHT)
    const next = drag.mode === 'move' ? moveRect(drag.origin, dx, dy) : resizeRect(drag.origin, dx, dy)
    setDraft({ id: drag.id, ...next })
  }

  const release = () => {
    if (drag && draft && draft.id === drag.id)
      update.mutate(
        { id: drag.id, changes: { x: draft.x, y: draft.y, width: draft.width, height: draft.height } },
        { onError: (error) => toast.error(error.message), onSettled: () => setDraft(null) },
      )
    else setDraft(null)
    setDrag(null)
  }

  return (
    <>
      <PageHeader title={t('title')} description={t('description', { items: tc('items', { count: all.length }) })}>
        <Button className="h-9 px-3" onClick={() => setAdding(true)}>
          <Plus />
          {t('add')}

            ref={canvas}
            aria-label={t('floor')}
            className="relative w-full touch-none overflow-hidden rounded-xl border bg-muted/40 select-none"
            style={{ aspectRatio: `${LAYOUT_WIDTH} / ${LAYOUT_HEIGHT}` }}
            onPointerMove={(event) => track(event as unknown as PointerEvent)}
            onPointerUp={release}
            onPointerCancel={release}
            onPointerDown={() => setSelectedId(null)}
          >
```

```tsx
function SplitDialog({ order, onClose }: { order: Order; onClose: () => void }) {
  const { t } = useTranslation('pos')
  const { t: tc } = useTranslation()
  const split = useSplitOrder()
  const workOnOrder = useCartStore((state) => state.workOnOrder)
  // How many of each line go to the new order.
  const [taken, setTaken] = useState<Record<number, number>>({})

  const lines = Object.entries(taken)
    .filter(([, quantity]) => quantity > 0)
    .map(([lineId, quantity]) => ({ lineId: Number(lineId), quantity }))
  const units = order.items.reduce((total, line) => total + line.quantity, 0)
  const takenUnits = lines.reduce((total, line) => total + line.quantity, 0)
  // At least one unit must stay on the original order.
  const canSplit = lines.length > 0 && takenUnits < units && !split.isPending

  const change = (lineId: number, quantity: number, max: number) =>
    setTaken({ ...taken, [lineId]: Math.min(Math.max(quantity, 0), max) })

  return (
    <Dialog open onOpenChange={(open) => !open && onClose()}>
      <DialogContent className="sm:max-w-md">
        <DialogHeader>
          <DialogTitle>{t('tools.split')}</DialogTitle>
          <DialogDescription>{t('tools.splitDescription')}</DialogDescription>
        </DialogHeader>
        <ul className="max-h-[50vh] divide-y overflow-y-auto rounded-lg border">
          {order.items.map((line) => (
            <li key={line.lineId} className="flex items-center justify-between gap-3 px-4 py-2">
              <span className="min-w-0 truncate">
                {line.productName}
                {line.options.length > 0 && <span className="text-xs text-muted-foreground"> · {line.options.map((option) => option.name).join(', ')}</span>}
              </span>
              <span className="flex shrink-0 items-center gap-2">
                <Button variant="outline" className="size-9" aria-label={t('tools.less', { name: line.productName })} onClick={() => change(line.lineId, (taken[line.lineId] ?? 0) - 1, line.quantity)}>
                  −
                </Button>
                <span className="w-14 text-center tabular-nums">{taken[line.lineId] ?? 0} / {line.quantity}</span>
                <Button variant="outline" className="size-9" aria-label={t('tools.more', { name: line.productName })} onClick={() => change(line.lineId, (taken[line.lineId] ?? 0) + 1, line.quantity)}>
                  +
                </Button>
              </span>
            </li>
          ))}
        </ul>
        {lines.length > 0 && takenUnits >= units && <p className="text-sm text-destructive">{t('tools.leaveOne')}</p>}
        <DialogFooter>
          <Button variant="outline" className="h-10" onClick={onClose}>
            {tc('cancel')}
          </Button>
          <Button
            className="h-10"
            disabled={!canSplit}
            onClick={() =>
              split.mutate(
                { orderId: order.id, lines },
                {
                  onSuccess: (created) => {
                    toast.success(t('tools.splitDone'))
                    workOnOrder(created.id)
                    onClose()
                  },
                  onError: (error) => toast.error(error.message),
                },
              )
            }
          >
            {t('tools.split')}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}
```
